const express = require('express'); const axios = require('axios'); const app = express(); app.use(express.json()); // Endpoint to trigger payment checkout app.post('/api/checkout', async (req, res) => { const { customerPhone } = req.body; const amount = "500.00"; // Request payload structure provided by official Merchant API documentation const payload = { merchantId: process.env.MERCHANT_ID, storeId: process.env.STORE_ID, transactionAmount: amount, customerMobileNumber: customerPhone, transactionType: "MWALLET", // Additional parameters and HMAC security signatures required by gateway }; try { // Send request to Official Gateway URL const gatewayResponse = await axios.post('https://sandbox.paymentgateway.com/api/v1/charge', payload, { headers: { 'Authorization': `Bearer ${process.env.API_TOKEN}`, 'Content-Type': 'application/json' } }); res.status(200).json({ success: true, status: gatewayResponse.data.status }); } catch (error) { res.status(500).json({ success: false, message: "Payment request failed to initialize." }); } }); app.listen(3000, () => console.log('Server running on port 3000'));
